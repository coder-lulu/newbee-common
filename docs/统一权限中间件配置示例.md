## 基础配置示例

### 🏢 企业级API服务配置

**企业管理系统 (enterprise.yaml)**:
```yaml
Name: enterprise-api
Host: 0.0.0.0
Port: 8080

# 数据库配置
DatabaseConf:
  Type: mysql
  Host: 192.168.1.100
  Port: 3306
  DBName: enterprise
  Username: app_user
  Password: "secure_password"

# Redis配置
RedisConf:
  Host: 192.168.1.101:6379
  Pass: "redis_password"
  Db: 0

# Core RPC配置
CoreRpc:
  Target: 192.168.1.102:9101
  Timeout: 5000

# 统一中间件配置
Middleware:
  # JWT认证配置
  auth:
    enabled: true
    accessSecret: "enterprise-jwt-secret-key-32-chars-min"
    accessExpire: 28800  # 8小时
    skipPaths:
      - "/health"
      - "/metrics"
      - "/api/v1/auth/login"
      - "/api/v1/auth/register"
      - "/api/v1/captcha"
      - "/swagger"

  # 租户检查配置  
  tenantCheck:
    enabled: true
    validateStatus: true
    cacheEnabled: true
    rateLimitEnabled: true
    maxRequestsPerMin: 1000
    skipPaths:
      - "/health"
      - "/api/v1/auth/login"
      - "/api/v1/captcha"

  # 数据权限配置
  dataPerm:
    enabled: true
    casbinEnabled: true
    coreRpcEndpoint: "192.168.1.102:9101"
    skipPaths:
      - "/health"
      - "/metrics"
      - "/api/v1/auth/login"
      - "/api/v1/captcha"
    
    # 统一数据权限高级配置
    unified:
      # 权限引擎配置
      engine:
        templatesEnabled: true
        dynamicRules: true
        priorityStrategy: "highest"
        customTemplates:
          - "/etc/config/templates/user.json"
          - "/etc/config/templates/department.json"
          - "/etc/config/templates/role.json"

      # 字段掩码配置
      fieldMask:
        enabled: true
        defaultStrategy: "partial"
        autoDetection: true
        accessStrategy: "role_based"
        customSensitive:
          # 身份信息
          "id_card": "hash"
          "passport_number": "hash"
          "social_security": "encrypt"
          
          # 联系方式
          "phone": "partial"
          "email": "partial"
          "home_address": "partial"
          
          # 认证信息
          "password": "full"
          "login_token": "full"
          "api_key": "encrypt"
          "secret_key": "encrypt"
          
          # 财务信息
          "salary": "encrypt"
          "bank_account": "hash"
          "credit_card": "hash"

      # 拦截器配置
      interceptor:
        sqlFiltering: true
        strictMode: true
        protectSystemTables: true
        skipTables:
          - "sys_audit_logs"
          - "sys_configurations"
          - "migrations"
          - "casbin_rules"

      # 性能配置
      performance:
        cacheEnabled: true
        cacheExpiry: 30        # 30分钟缓存
        batchConcurrency: 20
        checkTimeout: 3000     # 3秒超时
        metricsEnabled: true

      # 调试配置（生产环境关闭）
      debug:
        verboseLogging: false
        logPermissionChecks: false
        logSqlFilters: false
        logFieldMasks: false
        profileEnabled: false

  # 审计配置
  audit:
    enabled: true
    writerType: "rpc"
    maxBodySize: 2097152    # 2MB
    asyncEnabled: true
    asyncWorkers: 5
    asyncBufferSize: 2000
    skipPaths:
      - "/health"
      - "/metrics"

  # RBAC权限配置（可选）
  permission:
    enabled: false
    skipPaths:
      - "/health"
      - "/metrics"
```

### 配置验证脚本

**validate-config.sh**:
```bash
#!/bin/bash

echo "开始验证统一权限中间件配置..."

# 1. 检查必需的配置项
echo "检查基础配置..."
if ! grep -q "dataPerm:" config.yaml; then
    echo "❌ 缺少dataPerm配置"
    exit 1
fi

if ! grep -q "accessSecret:" config.yaml; then
    echo "❌ 缺少JWT密钥配置"
    exit 1
fi

# 2. 检查JWT密钥长度
SECRET_LENGTH=$(grep "accessSecret:" config.yaml | awk '{print $2}' | tr -d '"' | wc -c)
if [ $SECRET_LENGTH -lt 32 ]; then
    echo "❌ JWT密钥长度不足32位: $SECRET_LENGTH"
    exit 1
fi

# 3. 检查Redis连接
echo "检查Redis连接..."
REDIS_HOST=$(grep -A5 "RedisConf:" config.yaml | grep "Host:" | awk '{print $2}')
if ! redis-cli -h $REDIS_HOST ping > /dev/null 2>&1; then
    echo "❌ Redis连接失败: $REDIS_HOST"
    exit 1
fi

# 4. 检查Core RPC服务
echo "检查Core RPC服务..."
CORE_RPC=$(grep -A5 "CoreRpc:" config.yaml | grep "Target:" | awk '{print $2}')
if ! nc -z ${CORE_RPC%:*} ${CORE_RPC#*:} > /dev/null 2>&1; then
    echo "❌ Core RPC服务不可达: $CORE_RPC"
    exit 1
fi

echo "✅ 配置验证通过！"
```
